⚡ Web Builder Integration Example

Copy and paste this boilerplate into your HTML project: 

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>My Web Builder Integration</title>
</head>
<body style="margin: 0; padding: 0;">

    <div id="builder-container"></div>

    <script src="https://muhammadtaqi.oneapp.dev/web-builder.js"></script>

    <script>
        new MuhammadTaqiBuilder({
            containerId: "builder-container",
            builderName: "NEXURA TAQI"
        });
    </script>
</body>
</html>
